# Windows Service Configuration
# 使用 nssm (Non-Sucking Service Manager) 安装为 Windows 服务

# 安装命令示例:
# nssm install HslCommunicationService "C:\Program Files\dotnet\dotnet.exe" "C:\HslCommunication\HslCommunication.Service.dll"
# nssm set HslCommunicationService AppDirectory "C:\HslCommunication"
# nssm set HslCommunicationService DisplayName "HslCommunication Industrial Communication Service"
# nssm set HslCommunicationService Description "工业通信后台服务，支持多种PLC协议"
# nssm set HslCommunicationService Start SERVICE_AUTO_START
# nssm set HslCommunicationService ObjectName "NT AUTHORITY\LocalService"
# nssm set HslCommunicationService Environment "ASPNETCORE_ENVIRONMENT=Production" "ASPNETCORE_URLS=http://+:8080"

# 参数说明:
# - ServiceName: HslCommunicationService
# - DisplayName: HslCommunication 工业通信服务
# - Description: 工业通信后台服务，支持多种PLC协议
# - Start: SERVICE_AUTO_START (自动启动)
# - AppDirectory: 服务工作目录
# - ObjectName: 运行账户 (LocalService, NetworkService, 或域用户)
# - Environment: 环境变量

# 日志配置:
# nssm set HslCommunicationService AppStdout "C:\HslCommunication\logs\service.log"
# nssm set HslCommunicationService AppStderr "C:\HslCommunication\logs\error.log"
# nssm set HslCommunicationService AppRotateFiles 1
# nssm set HslCommunicationService AppRotateSeconds 86400
# nssm set HslCommunicationService AppRotateBytes 10485760

# 启动服务:
# net start HslCommunicationService

# 停止服务:
# net stop HslCommunicationService

# 卸载服务:
# nssm remove HslCommunicationService confirm
